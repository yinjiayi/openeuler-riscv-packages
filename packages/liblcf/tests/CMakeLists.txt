cmake_minimum_required(VERSION 3.16)
project(liblcf_installed_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(liblcf 0.8.1 EXACT CONFIG REQUIRED)
file(GLOB originals CONFIGURE_DEPENDS tests/*.cpp)
list(LENGTH originals original_count)
if(NOT original_count EQUAL 10)
  message(FATAL_ERROR "Expected all ten unchanged original test translation units")
endif()
add_executable(installed_originals ${originals})
target_compile_definitions(installed_originals PRIVATE DOCTEST_CONFIG_TREAT_CHAR_STAR_AS_STRING=1)
target_compile_options(installed_originals PRIVATE -UNDEBUG)
target_link_libraries(installed_originals PRIVATE liblcf::liblcf)
add_executable(installed_consumer consumer.cpp)
target_compile_options(installed_consumer PRIVATE -UNDEBUG)
target_link_libraries(installed_consumer PRIVATE liblcf::liblcf)
